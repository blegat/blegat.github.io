  \begin{frame}[fragile]
    \frametitle{MathOptInterface (MOI)}
    MOI in a nutshell:
    \begin{itemize}
      \item \verb|add_variable(model)|.
      \item \verb|add_constraint(model, func, set)|, e.g. $2x + 3y = 1$ $\to$ (\verb|2*x + 3*y|)-in-\verb|EqualTo(1.0)|.
      \item \verb|set|, \verb|get| attributes, e.g., \verb|ObjectiveSense|, \verb|ObjectiveFunction|.
    \end{itemize}
    Extensible framework:
    \begin{itemize}
      \item \alert{Generic} on attribute, function and set types. New ones can be defined \alert{independently}.
      \item Solver-\alert{specific} features easily exposed to JuMP/MOI users through \alert{custom} attributes.
      \item Expose \alert{specialized} problem structure easily through \alert{custom} functions, sets.
    \end{itemize}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Genericity and performance}
    What is the \alert{cost} of this \alert{genericity} ?

    Can MOI be both \alert{extensible} and \alert{performant} ?
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Matrix representation}
    Many \alert{LP} solvers use the following form:
\[
    \begin{array}{rl}
        \min\limits_{\mathbf{x}\in\mathbb{R}^N} & \mathbf{c}^\top \mathbf{x}\\
        \text{subject to:} & \mathbf{l} \le A\mathbf{x} \le \mathbf{u}\\
                           & \mathbf{x_l} \le \mathbf{x} \le \mathbf{x_u}
    \end{array}
\]
    Many \alert{conic} solvers use the following form:
\[
    \begin{array}{rl}
        \min\limits_{\mathbf{x}\in\mathbb{R}^N} & \mathbf{c}^\top \mathbf{x}\\
        \text{subject to:} & A\mathbf{x} + \mathbf{b} \in \mathcal{K}\\
                           & \mathbf{x}\ \mathsf{free}.
    \end{array}
    \qquad
    \begin{array}{rl}
        \max\limits_{\mathbf{x}\in\mathbb{R}^N} & -\mathbf{b}^\top \mathbf{y}\\
        \text{subject to:} & A^\Tr \mathbf{y} = \mathbf{c}\\
                           & \mathbf{y} \in \mathcal{K},
    \end{array}
\]
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Sparse matrices}
    $A$ is a \alert{sparse} $m \times n$ matrix of $k$ entries. Common formats:
    \begin{itemize}
      \item COO: 3 $k$-vectors: row indices, columns indices, entries (may be duplicates).
      \item CSC: 2 $k$-vectors: row indices, entries. 1 $(n+1)$-vector: column pointers.
      \item CSR: 2 $k$-vectors: column indices, entries. 1 $(n+1)$-vector: row pointers.
    \end{itemize}
    Need for \alert{conversion}:
    \begin{itemize}
      \item \texttt{MOI.VectorAffineFunction}: COO
      \item Multiple \texttt{MOI.ScalarAffineFunction}: CSR-like (not sorted, with duplicate)
      \item Most solvers: CSC
    \end{itemize}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Conversions}
    \begin{block}{COO $\to$ CSC}
      \begin{enumerate}
        \item \alert{Sort} and merge entries $\bigoh(k \log k)$ (may improve with bucket sort)
        \item \alert{Allocate} vectors $\bigoh(k + n)$
        \item \alert{Fill} vectors $\bigoh(k + n)$
      \end{enumerate}
    \end{block}
    \begin{block}{CSR $\to$ CSC (or CSC transpose)}
      \begin{enumerate}
        \item First \alert{pass}: \alert{Count} number of entries per columns $\bigoh(k)$
        \item \alert{Allocate} vectors $\bigoh(k + n)$
        \item Second \alert{pass}: \alert{Fill} vectors $\bigoh(k + n)$
      \end{enumerate}
    \end{block}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Canonical functions}
    \begin{itemize}
      \item \texttt{is\_canonical}: $\bigoh(k)$ time, 0 allocations.
      \item \texttt{canonical}: $\bigoh(k\log k)$ time, $\bigoh(k)$ allocations.
    \end{itemize}
    \begin{itemize}
      \item First and second pass need \alert{canonical} MOI functions, i.e., sorted and without duplicates.
      \item JuMP creates \alert{canonical} functions.
      \item Bridges may \alert{break} canonicalization.
      \item \texttt{MOI.modify} may \alert{break} canonicalization.
    \end{itemize}

    \begin{itemize}
      \item \texttt{MOI.Utilities.Model} need to \alert{copy} function given by user: might as well canonicalize them as well.
      \item So \texttt{MOI.Utilities.Model} functions are \alert{canonical} unless \texttt{MOI.modify}'d
      \item \texttt{CanonicalConstraintFunction} check \texttt{is\_canonical} and canonicalize only if needed.
    \end{itemize}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Copy without bridges}
    \texttt{VectorOfConstraints} to \texttt{MatrixOfConstraints}
    \begin{enumerate}
      \item First pass: Collect functions with \texttt{CanonicalConstraintFunction}
        and \alert{count} number of entries by column: no allocation if \texttt{modify} not called. Function not mapped with \texttt{IndexMap} to avoid allocation.
      \item \alert{Allocate} CSC representation.
      \item Second pass: Go through collected function and
        \alert{fill} CSC representation.
    \end{enumerate}
    \begin{itemize}
      \item Canonicalization done \alert{once} as functions are collected.
      \item Indices are mapped \alert{twice} (once per pass) but mapped function not allocated and mapping is \alert{cheap} (\texttt{CleverDicts} is a \texttt{Vector} if no variable were deleted).
    \end{itemize}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Copy with bridges}
    \begin{itemize}
      \item All constraint types not bridged passed \alert{through} bridge layers with \texttt{pass\_nonvariable\_constraints} to avoid \alert{allocating} mapped functions.
      \item Bridged constraints add function with \alert{mapped} indices that need to be canonicalized.
      \item \alert{Mix} of functions with indices mapped or not mapped.
    \end{itemize}
    \begin{itemize}
      \item The bridge layer still \alert{allows} performance optimization.
      \item The bridging of some constraints types does \alert{not affect} performance
        of other constraint types.
    \end{itemize}
  \end{frame}

  \begin{frame}[fragile]
    \frametitle{Only two copies}
    The number of nonzero entries $k$ might be \alert{large}: we don't want to allocate to many \alert{copies}.
    \begin{itemize}
      \item \texttt{CachingOptimizer}: One copy of \alert{user} model in \texttt{VectorOfConstraints} form.
      \item \texttt{LazyBridgeOptimizer}: Bridged layer.
      \item \texttt{CachingOptimizer}: One copy of \alert{solver} model in \texttt{MatrixOfConstraints} form (0-based indexed if needed).
      \item Solver.
    \end{itemize}
    Copy of solver model is solver \alert{specific}: depends on supported constraints, 0-based vs 1-based, splitting of matrices between constraint types...
  \end{frame}
