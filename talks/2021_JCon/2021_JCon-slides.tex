\begin{frame}[fragile]
  \frametitle{Mutable sum: \texttt{BigInt}}
  Default implementation (simplified):
\begin{minted}{Julia}
function sum(x)
    acc = zero(eltype(x))
    for el in x
        acc = acc + el
    end
    return acc
end
\end{minted}
\begin{minted}{Julia}
x = big.([2, 0, 2, 1])
\end{minted}
  What is \texttt{sum(x)} doing ?

\begin{uncoverenv}<2>
  \sout{\texttt{acc = acc + el}} $\to$
  \texttt{Base.GMP.MPZ.add!(acc, el)}
\end{uncoverenv}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Mutable sum: first solution}
\begin{minted}{Julia}
function sum(x)
    acc = zero(eltype(x))
    for el in x
        acc = MA.add!(acc, el)
    end
    return acc
end
MA.add!(a, b) = a + b # default fallback
MA.add!(a::BigInt, b::BigInt) = Base.GMP.MPZ.add!(a, b)
function MA.add!(s::Sum, t::Term)
    push!(s.terms, t)
    return s
end
\end{minted}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Problem with first solution}
\begin{minted}{Julia}
struct Rational{T}
    num::T
    den::T
end
function MA.mul!(a::Rational, b::Rational)
    if # T is mutable
        MA.mul!(a.num, b.num)
        MA.mul!(a.den, b.den)
        # ... simplify with gcd ...
        return a
    else
        return a * b
    end
end
\end{minted}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Explicit mutability}
\begin{minted}{Julia}
MA.mul!(a, b) = MA.operate!(*, a, b)
function MA.operate!(op, a::A, b::B) where {A, B}
    if MA.mutability(A, op, A, B) isa MA.IsMutable
        return MA.mutable_operate!(op, a, b)
    else
        return op(a, b)
    end
end
function MA.operate_to!(out::O, op, a::A, b::B) where ...
    if MA.mutability(O, op, A, B) isa MA.IsMutable
        return MA.mutable_operate_to!(out, op, a, b)
    else
        return op(a, b)
    end
end
\end{minted}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Mutable API}
\texttt{MA.operate!}/\texttt{MA.operate\_to!} \alert{may} mutate or not first argument.
Returned value is \alert{always} the result.

\texttt{MA.mutable\_operate!}/\texttt{MA.mutable\_operate\_to!} mutates or \alert{errors} (default).
First argument is \alert{always} the result.

  \begin{block}{Axiom of mutable arithmetics}
\begin{minted}{Julia}
function mutability(T::Type, op, args...)
    if mutability(T) isa IsMutable &&
        promote_operation(op, args...) == T
        return IsMutable()
    else
        return NotMutable()
    end
end
\end{minted}
  \end{block}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Implementing the API}
  \texttt{mutability(T)} defaults to \texttt{NotMutable()}.
  Hence, by default,
  \texttt{MA.operate!(op, args...)} redirects to \texttt{op(args...)}.

  \begin{block}{Flipping the switch}
  After implementing \texttt{mutability(::Type{T}) = IsMutable()},
  unimplemented \texttt{mutable\_operate!}/\texttt{mutable\_operate\_to!}
  gives helpful errors.

  \alert{No silent} fallback to inefficient \texttt{op(args...)}!

  \texttt{promote\_operation} falls back to \texttt{typeof(op(zero.(args)...))}.

  \end{block}
  More \alert{features} for the user with an API \alert{easier} to implement.
\end{frame}

\begin{frame}[fragile]
  \frametitle{Rewriting macro}
  \begin{minted}{Julia}
MA.@rewrite(x + y * z - a * b * c) # rewritten into:
v1 = MA.copy_if_mutable(x)
v2 = MA.operate!(MA.add_mul, v1, y, z)
v3 = MA.operate!(MA.sub_mul, v2, a, b, c)
  \end{minted}
  Used in JuMP macros, critical for modeling performance.
  \begin{minted}{Julia}
MA.add_mul(x, args...) = x + *(args...)
MA.sub_mul(x, args...) = x - *(args...)
function MA.copy_if_mutable(x)
    if MA.mutability(x) isa MA.IsMutable
        return MA.mutable_copy(x)
    end
    return x
end
  \end{minted}
\end{frame}
